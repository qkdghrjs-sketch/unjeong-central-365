  var rvs = root.querySelectorAll('.eum_rv');
  if (reduce || !window.IntersectionObserver) {
    Array.prototype.forEach.call(rvs, function (el) { el.classList.add('eum_in'); });
  } else {
    var obs = new IntersectionObserver(function (es) {
      es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('eum_in'); obs.unobserve(e.target); } });
    }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });
    Array.prototype.forEach.call(rvs, function (el) { obs.observe(el); });
  }

  /* 커서를 따라오는 사진 — 마우스가 있는 넓은 화면에서만 */
  var peek = root.querySelector('.eum_peek');
  var shots = peek ? peek.querySelectorAll('img') : [];
  var rows = root.querySelectorAll('.eum_row');
  if (!peek || !rows.length) return;
  if (reduce || !window.matchMedia('(hover:hover) and (pointer:fine)').matches) return;

  var tx = 0, ty = 0, cx = 0, cy = 0, active = false, raf = null;

  function place() {
    peek.style.setProperty(
      'transform',
      'translate(' + Math.round(cx) + 'px,' + Math.round(cy) + 'px) translate(-50%,-50%) scale(' + (active ? 1 : 0.92) + ')',
      'important'
    );
  }

  function paint() {
    /* 커서 쪽으로 조금씩 좁혀가며 부드럽게 따라옵니다 */
    cx += (tx - cx) * 0.14;
    cy += (ty - cy) * 0.14;
    place();
    if (active || Math.abs(tx - cx) > 0.5 || Math.abs(ty - cy) > 0.5) {
      raf = requestAnimationFrame(paint);
    } else {
      raf = null;
    }
  }

  function onMove(e) {
    tx = e.clientX;
    ty = e.clientY;
    if (!raf) raf = requestAnimationFrame(paint);
  }

  Array.prototype.forEach.call(rows, function (row) {
    var i = parseInt(row.getAttribute('data-i'), 10);
    row.addEventListener('mouseenter', function (e) {
      if (window.innerWidth <= 860) return;
      tx = e.clientX; ty = e.clientY;
      if (!active) { cx = tx; cy = ty; }
      active = true;
      peek.classList.add('eum_on');
      for (var n = 0; n < shots.length; n++) {
        shots[n].classList[n === i ? 'add' : 'remove']('eum_on');
      }
      place();   /* 첫 프레임부터 커서 자리에 놓습니다 */
      if (!raf) raf = requestAnimationFrame(paint);
    });
    row.addEventListener('mousemove', onMove);
    row.addEventListener('mouseleave', function () {
      active = false;
      peek.classList.remove('eum_on');
    });
  });

  /* 행 사이 여백에서도 자연스럽게 따라오도록 */
  Array.prototype.forEach.call(root.querySelectorAll('.eum_rows'), function (list) {
    list.addEventListener('mousemove', onMove);
  });
})();
</script>
