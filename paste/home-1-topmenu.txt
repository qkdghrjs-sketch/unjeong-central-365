(function () {
  var root = document.getElementById('eumHd');
  if (!root || root.dataset.eumReady) return;   /* 아임웹에서 두 번 실행되는 것 방지 */
  root.dataset.eumReady = '1';

  /* 스크롤하면 흰 배경 */
  function onScroll() {
    if (window.pageYOffset > 40) root.classList.add('eum_solid');
    else root.classList.remove('eum_solid');
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  /* PC 전체메뉴 — 마우스를 올리면 세부 페이지가 펼쳐집니다 */
  var gnbItems = root.querySelectorAll('.eum_gnb > li');
  var cols = root.querySelectorAll('.eum_mcol');
  var timer = null;

  function hot(g) {
    for (var i = 0; i < gnbItems.length; i++) {
      gnbItems[i].classList[gnbItems[i].getAttribute('data-g') === g ? 'add' : 'remove']('eum_hot');
    }
    for (var j = 0; j < cols.length; j++) {
      cols[j].classList[cols[j].getAttribute('data-g') === g ? 'add' : 'remove']('eum_hot');
    }
  }
  function open(g) {
    if (window.innerWidth <= 1024) return;
    clearTimeout(timer);
    root.classList.add('eum_open');
    hot(g);
  }
  function close(delay) {
    clearTimeout(timer);
    timer = setTimeout(function () {
      root.classList.remove('eum_open');
      hot(null);
    }, delay === undefined ? 140 : delay);
  }

  Array.prototype.forEach.call(gnbItems, function (li) {
    var g = li.getAttribute('data-g');
    li.addEventListener('mouseenter', function () { open(g); });
    var a = li.querySelector('a');
    if (a) a.addEventListener('focus', function () { open(g); });
  });
  Array.prototype.forEach.call(cols, function (c) {
    c.addEventListener('mouseenter', function () { open(c.getAttribute('data-g')); });
  });
  root.addEventListener('mouseenter', function () { clearTimeout(timer); });
  root.addEventListener('mouseleave', function () { close(); });
  document.addEventListener('keydown', function (e) { if (e.key === 'Escape') close(0); });

  /* 모바일 메뉴 + 아코디언 */
  var burger = root.querySelector('.eum_burger');
  var groups = root.querySelectorAll('.eum_mg');

  burger.addEventListener('click', function () {
    var on = root.classList.toggle('eum_mo');
    burger.setAttribute('aria-expanded', on ? 'true' : 'false');
    burger.setAttribute('aria-label', on ? '메뉴 닫기' : '메뉴 열기');
    document.documentElement.style.overflow = on ? 'hidden' : '';
    if (on && groups.length && !root.querySelector('.eum_mg.eum_on')) groups[0].classList.add('eum_on');
  });

  Array.prototype.forEach.call(groups, function (g) {
    g.querySelector('.eum_mh').addEventListener('click', function () {
      var was = g.classList.contains('eum_on');
      Array.prototype.forEach.call(groups, function (x) { x.classList.remove('eum_on'); });
      if (!was) g.classList.add('eum_on');
    });
  });

  Array.prototype.forEach.call(root.querySelectorAll('.eum_mm a'), function (a) {
    a.addEventListener('click', function () {
      root.classList.remove('eum_mo');
      document.documentElement.style.overflow = '';
      burger.setAttribute('aria-expanded', 'false');
    });
  });
})();
</script>
