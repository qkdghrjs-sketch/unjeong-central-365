<script>
(function () {
    function initEumHeader() {
        var headerWrap = document.getElementById('eum_header_wrap');
        if (!headerWrap || headerWrap.getAttribute('data-eum-ready') === '1') return;
        headerWrap.setAttribute('data-eum-ready', '1');
        var hamburger = document.getElementById('eum_hamburger');
        var closeBtn  = document.getElementById('eum_m_close');
        var mobileNav = document.getElementById('eum_mobile_nav');
        var overlay   = document.getElementById('eum_mobile_overlay');
        var bodyOverflowBeforeOpen = '';
        function syncScrolledState() {
            if (window.scrollY > 10) {
                headerWrap.classList.add('eum_scrolled');
            } else {
                headerWrap.classList.remove('eum_scrolled');
            }
        }
        window.addEventListener('scroll', syncScrolledState, { passive: true });
        syncScrolledState();
        function openMenu() {
            if (!mobileNav || !overlay) return;
            bodyOverflowBeforeOpen = document.body.style.overflow || '';
            mobileNav.classList.add('eum_active');
            overlay.classList.add('eum_active');
            mobileNav.setAttribute('aria-hidden', 'false');
            if (hamburger) hamburger.setAttribute('aria-expanded', 'true');
            document.body.style.overflow = 'hidden';
        }
        function closeMenu() {
            if (!mobileNav || !overlay) return;
            mobileNav.classList.remove('eum_active');
            overlay.classList.remove('eum_active');
            mobileNav.setAttribute('aria-hidden', 'true');
            if (hamburger) hamburger.setAttribute('aria-expanded', 'false');
            document.body.style.overflow = bodyOverflowBeforeOpen;
        }
        if (hamburger) hamburger.addEventListener('click', openMenu);
        if (closeBtn) closeBtn.addEventListener('click', closeMenu);
        if (overlay) overlay.addEventListener('click', closeMenu);
        document.addEventListener('keydown', function (e) {
            if (e.key === 'Escape' && mobileNav && mobileNav.classList.contains('eum_active')) {
                closeMenu();
            }
        });
        var hasSubs = headerWrap.querySelectorAll('.eum_m_has_sub');
        function isDummyHref(href) {
            href = (href || '').trim().toLowerCase();
            return href === '' || href === '#' || href === '#none' || href.indexOf('javascript:void') === 0;
        }
        Array.prototype.forEach.call(hasSubs, function (item) {
            var link = item.children[0];
            var lnb  = item.querySelector('.eum_m_lnb');
            if (!link || !lnb) return;
            link.addEventListener('click', function (e) {
                var rawHref = link.getAttribute('href') || '';
                var isOpen = item.classList.contains('eum_open');
                if (isOpen && !isDummyHref(rawHref)) return true;
                e.preventDefault();
                if (isOpen) {
                    item.classList.remove('eum_open');
                    lnb.style.maxHeight = null;
                    return false;
                }
                Array.prototype.forEach.call(hasSubs, function (other) {
                    other.classList.remove('eum_open');
                    var otherLnb = other.querySelector('.eum_m_lnb');
                    if (otherLnb) otherLnb.style.maxHeight = null;
                });
                item.classList.add('eum_open');
                lnb.style.maxHeight = lnb.scrollHeight + 'px';
                return false;
            });
        });
        /* 신규 메뉴 링크 주소가 아직 없을 때 #none 클릭으로 페이지가 튀는 현상을 방지 */
        Array.prototype.forEach.call(headerWrap.querySelectorAll('a[href="#none"], a[href="#"]'), function (link) {
            link.addEventListener('click', function (e) {
                if (!link.closest('.eum_m_has_sub') || link.parentElement.classList.contains('eum_m_lnb_inner')) {
                    e.preventDefault();
                }
            });
        });
        function killImwebDefaults() {
            if (window.innerWidth <= 1200) {
                var selectors = [
                    '#mobile_header_wrap', '#mobile_header', '#doz_header_wrap', '#doz_header',
                    '.mobile_menu_wrap', '.side_menu_wrap', '.widget_sitemap', '.widget_menu.doz_sys',
                    '.category-nav', '.visual_menu_wrap', '.sitemap_wrap', '.local_nav_wrap',
                    '.local_nav', '.tab_menu_wrap', '.sub_menu_wrap', '.site_map_list',
                    '.widget_menu_local', '.local_sub_menu', '.cm-drop-menu', '.mobile_bottom_menu',
                    '#mobile_bottom_menu'
                ];
                selectors.forEach(function (sel) {
                    document.querySelectorAll(sel).forEach(function (el) {
                        if (el.closest && el.closest('#eum_header_wrap')) return;
                        el.style.setProperty('display', 'none', 'important');
                        el.style.setProperty('visibility', 'hidden', 'important');
                        el.style.setProperty('opacity', '0', 'important');
                        el.style.setProperty('height', '0', 'important');
                        el.style.setProperty('pointer-events', 'none', 'important');
                    });
                });
            }
        }
        killImwebDefaults();
        var cnt = 0;
        var timer = setInterval(function () {
            killImwebDefaults();
            if (++cnt > 50) clearInterval(timer);
        }, 100);
        window.addEventListener('resize', function () {
            killImwebDefaults();
            if (window.innerWidth > 1200 && mobileNav && mobileNav.classList.contains('eum_active')) {
                closeMenu();
            }
        });
    }
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initEumHeader);
    } else {
        initEumHeader();
    }
})();
