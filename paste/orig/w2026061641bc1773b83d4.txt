    function eumOpenModal(type) {
        const data = eumModalData[type];
        const overlay = document.getElementById('eum_modal_overlay');
        const title = document.getElementById('eum_modal_title');
        const body = document.getElementById('eum_modal_body');
        const closeButton = document.getElementById('eum_modal_close');
        if (!data || !overlay || !title || !body) {
            return;
        }
        eumPreviousFocus = document.activeElement;
        title.textContent = data.title;
        body.innerHTML = data.content;
        body.scrollTop = 0;
        overlay.classList.add('eum_modal_active');
        overlay.setAttribute('aria-hidden', 'false');
        document.body.classList.add('eum_modal_body_locked');
        window.setTimeout(function () {
            if (closeButton) {
                closeButton.focus();
            }
        }, 30);
    }
    function eumCloseModal() {
        const overlay = document.getElementById('eum_modal_overlay');
        if (!overlay) {
            return;
        }
        overlay.classList.remove('eum_modal_active');
        overlay.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('eum_modal_body_locked');
        if (eumPreviousFocus && typeof eumPreviousFocus.focus === 'function') {
            eumPreviousFocus.focus();
        }
    }
    function eumInitModal() {
        const overlay = document.getElementById('eum_modal_overlay');
        const closeButton = document.getElementById('eum_modal_close');
        document.querySelectorAll('[data-eum-modal]').forEach(function (button) {
            button.addEventListener('click', function () {
                eumOpenModal(button.getAttribute('data-eum-modal'));
            });
        });
        if (closeButton) {
            closeButton.addEventListener('click', eumCloseModal);
        }
        if (overlay) {
            overlay.addEventListener('click', function (event) {
                if (event.target === overlay) {
                    eumCloseModal();
                }
            });
        }
        document.addEventListener('keydown', function (event) {
            if (event.key === 'Escape' && overlay && overlay.classList.contains('eum_modal_active')) {
                eumCloseModal();
            }
        });
        window.eumOpenModal = eumOpenModal;
        window.eumCloseModal = eumCloseModal;
    }
