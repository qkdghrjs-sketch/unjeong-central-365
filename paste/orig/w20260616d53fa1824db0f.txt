<script charset="UTF-8">
(function () {
    var rendered = false;
    function renderMap() {
        if (rendered) return;
        var container = document.getElementById('daumRoughmapContainer1781595868311');
        if (!container) return;
        try {
            container.innerHTML = '';
            new daum.roughmap.Lander({
                "timestamp" : "1781595868311",
                "key"       : "pmuctu7j989",
                "mapWidth"  : "100%",
                "mapHeight" : "100%"
            }).render();
            rendered = true;
            setTimeout(function () { window.dispatchEvent(new Event('resize')); }, 300);
            setTimeout(function () { window.dispatchEvent(new Event('resize')); }, 1000);
        } catch (e) { console.warn('Map render error:', e); }
    }
    if (document.readyState === 'complete') { setTimeout(renderMap, 500); }
    else { window.addEventListener('load', function () { setTimeout(renderMap, 500); }); }

    var rt;
    window.addEventListener('resize', function () {
        clearTimeout(rt);
        rt = setTimeout(function () {
            var mw = document.querySelector('#eum_loc_box .wrap_map');
            if (mw) mw.style.width = '100%';
        }, 200);
    });
})();
</script>
