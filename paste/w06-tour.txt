<script>
(function () {
  var root = document.getElementById('eumEnv');
  if (!root || root.dataset.eumReady) return;
  root.dataset.eumReady = '1';

  var els = root.querySelectorAll('.eum_rv');
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches || !window.IntersectionObserver) {
    Array.prototype.forEach.call(els, function (el) { el.classList.add('eum_in'); });
  } else {
    var obs = new IntersectionObserver(function (es) {
      es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('eum_in'); obs.unobserve(e.target); } });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    Array.prototype.forEach.call(els, function (el) { obs.observe(el); });
  }

  var rail = root.querySelector('.eum_rail');
  var fill = root.querySelector('.eum_fill');
  function paint() {
    var max = rail.scrollWidth - rail.clientWidth;
    var w = Math.max((rail.clientWidth / rail.scrollWidth) * 100, 12);
    var p = max > 0 ? rail.scrollLeft / max : 0;
    fill.style.width = w + '%';
    fill.style.left = (p * (100 - w)) + '%';
  }
  rail.addEventListener('scroll', paint, { passive: true });
  window.addEventListener('resize', paint);
  paint();
})();
</script>
