(function () {
  var root = document.getElementById('eumPhi');
  if (!root || root.dataset.eumReady) return;
  root.dataset.eumReady = '1';

  var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  /* 등장 */
  var rvs = root.querySelectorAll('.eum_rv');
  if (reduce || !window.IntersectionObserver) {
    Array.prototype.forEach.call(rvs, function (el) { el.classList.add('eum_in'); });
  } else {
    var obs = new IntersectionObserver(function (es) {
      es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('eum_in'); obs.unobserve(e.target); } });
    }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });
    Array.prototype.forEach.call(rvs, function (el) { obs.observe(el); });
  }

  /* 스크롤에 따라 사진·숫자가 저절로 넘어갑니다 (클릭 없음) */
  var items = root.querySelectorAll('.eum_item');
  var shots = root.querySelectorAll('.eum_stage img');
  var big   = root.querySelector('.eum_big');
  var fill  = root.querySelector('.eum_fill');
  var cnt   = root.querySelector('.eum_count b');
  if (!items.length || !shots.length) return;

  var cur = -1;
  function setActive(i) {
    if (i === cur || i < 0 || i >= items.length) return;
    cur = i;
    for (var n = 0; n < items.length; n++) items[n].classList[n === i ? 'add' : 'remove']('eum_on');
    for (var m = 0; m < shots.length; m++) shots[m].classList[m === i ? 'add' : 'remove']('eum_on');
    var label = ('0' + (i + 1)).slice(-2);
    if (big) big.textContent = label;
    if (cnt) cnt.textContent = label;
    if (fill) fill.style.setProperty('width', ((i + 1) / items.length * 100) + '%', 'important');
  }

  /* 화면 한가운데에 가장 가까운 항목이 활성화됩니다 */
  var ticking = false;
  function update() {
    ticking = false;
    var mid = window.innerHeight * 0.5;
    var best = 0, bestDist = Infinity;
    for (var i = 0; i < items.length; i++) {
      var r = items[i].getBoundingClientRect();
      var d = Math.abs(r.top + r.height / 2 - mid);
      if (d < bestDist) { bestDist = d; best = i; }
    }
    setActive(best);
  }
  function onScroll() {
    if (ticking) return;
    ticking = true;
    if (window.requestAnimationFrame) requestAnimationFrame(update);
    else setTimeout(update, 60);
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  window.addEventListener('resize', onScroll);
  update();
})();
</script>
